using System;
using System.IO;
using System.Linq;
using System.Collections;
using System.Collections.Generic;
using System.Runtime.Serialization.Json;
using BibleSurvivor.Core;
using BibleSurvivor.Unity;
using UnityEngine;
public sealed class R02ArtCapture:MonoBehaviour
{
    AdamGame game;AdamHost host;string root;bool draw;
    readonly List<string> facts=new List<string>();
    string Arg(string key){var a=Environment.GetCommandLineArgs();var i=Array.IndexOf(a,key);return i<0||i+1>=a.Length?null:a[i+1];}
    void Awake(){root=Arg("-r02ArtCaptureRoot");if(root==null){enabled=false;return;}if(!Path.IsPathRooted(root)||!File.Exists(Path.Combine(root,".adam-r02-art-qa")))throw new Exception("Fresh R02 art capture marker required");Application.runInBackground=true;}
    void Require(bool value,string message){if(!value)throw new Exception(message);}
    void Save(string name,object value){using(var s=File.Create(Path.Combine(root,name)))new DataContractJsonSerializer(value.GetType()).WriteObject(s,value);}
    IEnumerator Start(){if(!enabled)yield break;var steps=Run();while(true){var more=false;object item=null;try{more=steps.MoveNext();if(more)item=steps.Current;}catch(Exception e){Debug.LogException(e);Application.Quit(2);}if(!more)yield break;yield return item;}}
    void Install(AdamHost value,bool fx=true){host=value;game.Initialize(host);game.SetLifecycle(true,false);game.ArtWorld.FeedbackEnabled=fx;game.ImpactParticles.Enabled=true;game.ImpactAudio.Muted=true;host.SetPaused(false);game.UpdatePresentation(0);}
    AdamHost Fixture(string file,string weapon,int population,double time=80,string isolate=null,bool left=false)
    {
        var catalog=AdamCatalog.Load();var store=new SaveStore(file);var h=new AdamHost(catalog,store);h.Start(weapon);h.Session.Advance(180);h.SetPaused(true);h.SetPaused(false);
        for(var pass=0;pass<8;pass++)
        {
            h.Session.Advance(pass==0?time:0);if(h.Session.Wave.Active.Values.Any(x=>x.MonsterId=="b2"))break;
            var keep=h.Session.Wave.Active.Values.GroupBy(x=>x.MonsterId).Select(g=>g.First().Id).ToArray();foreach(var id in h.Session.Wave.Active.Keys.Where(x=>!keep.Contains(x)).ToArray())h.Session.Kill(id);
        }
        while(h.Session.Wave.Active.Count>population){var id=h.Session.Wave.Active.Values.GroupBy(x=>x.MonsterId).OrderByDescending(g=>g.Count()).First().Last().Id;h.Session.Kill(id);}
        h.Session.Capture();var state=h.State;var world=new BattleWorld(h.Session,catalog.Speeds,7).Capture();var i=0;
        var target=isolate==null?0:h.Session.Wave.Active.Values.First(x=>x.MonsterId==isolate).Id;
        foreach(var e in world.Enemies)
        {
            if(isolate!=null){e.X=e.Id==target?(left?3:-3):100;e.Y=e.Id==target?1:100;}
            else{e.X=(i%10-5)*.6f;e.Y=(i/10-2)*.65f;i++;}
        }
        foreach(var id in h.Session.DroppedExperience.Keys)world.Drops.Add(new BattleEntityState{Id=id,X=100,Y=100});state.Run.World=world;store.Commit(state,state.Revision,Guid.NewGuid().ToString("N"));Require(h.Reload(),h.Error);return h;
    }
    IEnumerator Resize(int width,int height){Screen.SetResolution(width,height,false);for(var i=0;i<90&&(Screen.width!=width||Screen.height!=height);i++)yield return null;Require(Screen.width==width&&Screen.height==height,"Resolution mismatch");game.SetLifecycle(true,false);}
    IEnumerator Shot(string name){game.UpdatePresentation(0);yield return new WaitForEndOfFrame();ScreenCapture.CaptureScreenshot(Path.Combine(root,name+".png"));yield return null;}
    IEnumerator Run()
    {
        while((game=UnityEngine.Object.FindFirstObjectByType<AdamGame>())==null||game.Host==null)yield return null;
        game.enabled=false;game.Initialize(game.Host);draw=true;
        var persistent=Path.Combine(Application.persistentDataPath,"adam-prototype.json");
        if(Arg("-r02ArtReload")!=null)
        {
            host=game.Host;Require(host.Session.RoundNumber==2&&host.Paused,"Reload R02 paused");Save("loaded-state.json",host.State);Save("loaded-world.json",host.World.Capture());yield return Shot("loaded-r02");Application.Quit(0);yield break;
        }
        var art=new AdamR02Art();
        // Opaque source pixels must be present at the actual product's integer registration, including UV reflection.
        foreach(var size in new[]{660,1100})
        {
            yield return Resize(size,size==660?440:720);
            foreach(var entry in art.Index.assets)foreach(var left in new[]{false,true})
            for(var frame=0;frame<(size==660?4:1);frame++)
            {
                var name="probe-"+size+"-"+entry.monster+"-"+(left?"left":"right")+"-"+frame;
                Install(Fixture(Path.Combine(root,name+".json"),"sling",50,80+frame/(double)entry.clip.fps,entry.monster,left));
                Require(entry.clip.Frame(game.ArtWorld.Clock)==frame,"Probe clock mismatch");
                for(var settle=0;settle<3;settle++)yield return null;yield return new WaitForEndOfFrame();var capture=ScreenCapture.CaptureScreenshotAsTexture();
                var pixels=capture.GetPixels32();var sprite=art.Sprite(entry.clip.id,frame);var source=sprite.texture.GetPixels32();
                var arena=AdamGame.Arena(Screen.width,Screen.height);var projection=new AdamWorldProjection(arena,Vector2.zero);var feet=projection.Point(left?3:-3,1)+arena.position;var rect=AdamArt.PlacedRect(entry.clip,feet,projection.Scale);var checkedPixels=0;var matched=0;var maxDelta=0;
                for(var y=0;y<32;y++)for(var x=0;x<32;x++)
                {
                    var expected=source[y*128+frame*32+x];if(expected.a!=255)continue;
                    // Compare display-encoded framebuffer values; imported raw RGBA is separately hash-validated.
                    if(QualitySettings.activeColorSpace==ColorSpace.Linear)expected=(Color32)((Color)expected).gamma;
                    var tx=Mathf.FloorToInt(rect.x+(left?31-x:x)*projection.Scale+projection.Scale/2f);var ty=Screen.height-1-Mathf.FloorToInt(rect.y+(31-y)*projection.Scale+projection.Scale/2f);
                    var actual=pixels[ty*Screen.width+tx];var delta=Math.Max(Math.Abs(actual.r-expected.r),Math.Max(Math.Abs(actual.g-expected.g),Math.Abs(actual.b-expected.b)));maxDelta=Math.Max(maxDelta,delta);checkedPixels++;if(delta<=5)matched++; // Repeated native readback differed by up to4 display code values; spatial/frame checks remain exact.
                }
                File.WriteAllBytes(Path.Combine(root,name+".png"),capture.EncodeToPNG());UnityEngine.Object.Destroy(capture);facts.Add(name+" opaque="+checkedPixels+" matched="+matched+" scale="+projection.Scale+" frame="+frame+" colorSpace="+QualitySettings.activeColorSpace+" maxRgbDelta="+maxDelta);File.WriteAllLines(Path.Combine(root,"pixel-probes.txt"),facts);
                Require(checkedPixels>100 && matched>=checkedPixels*.98,"Actual R02 pixels missing/wrong frame/flip: "+facts.Last());yield return null;
            }
        }
        foreach(var size in new[]{660,1100})foreach(var count in new[]{40,50})foreach(var weapon in AdamGame.Weapons)foreach(var fx in new[]{false,true})
        {
            yield return Resize(size,size==660?440:720);var name="mixed-"+size+"-"+count+"-"+weapon+"-"+(fx?"normal":"low");Install(Fixture(Path.Combine(root,name+".json"),weapon,count),fx);
            Require(host.Session.Wave.Active.Count==count,"Exact initial population");yield return Shot(name+"-start");var peak=0;
            for(var tick=0;tick<120 && host.Session.IsActive;tick++)
            {
                while(host.Session.PendingChoices.Count>0)Require(host.Choose(host.Session.PendingChoices[0]),host.Error);
                host.SetMovement(tick<60?Vector2.right:Vector2.left);host.Tick(1d/60);game.UpdatePresentation(1f/60);peak=Math.Max(peak,game.ImpactParticles.ParticleCount);Require(peak<=256,"Particle pool overflow");
                if(tick==20||tick==80)yield return Shot(name+"-"+tick);
            }
            if(!fx)Require(game.ImpactParticles.ParticleCount==0,"LowFX kept particles");host.SetPaused(true);var time=host.World.SimulationTime;var clock=game.ArtWorld.Clock;yield return Shot(name+"-pause");yield return new WaitForSecondsRealtime(.08f);game.UpdatePresentation(.1f);Require(time==host.World.SimulationTime&&clock==game.ArtWorld.Clock,"Pause changed battle/animation time");
            facts.Add(name+" initial="+count+" hits="+host.World.Hits+" kills="+host.World.Kills+" peakParticles="+peak+" pauseClock="+clock);File.WriteAllLines(Path.Combine(root,"matrix.txt"),facts.Where(x=>x.StartsWith("mixed-")));
        }
        Install(Fixture(persistent,"sling",50));host.Tick(.1);host.SetPaused(true);Require(host.Checkpoint(),host.Error);Save("persisted-state.json",host.State);Save("persisted-world.json",host.World.Capture());yield return Shot("persisted-r02-paused");File.WriteAllText(Path.Combine(root,"complete.txt"),"Actual product 30 frame/flip pixel probes + 24 mixed population/weapon/FX/resolution cases; explicit staged fixtures; not natural manual balance/performance/mobile");Application.Quit(0);
    }
    void OnGUI(){if(draw&&game!=null)game.DrawGameView(AdamGame.ScreenViewport());}
}
